cmake_minimum_required(VERSION 3.24)
project(Stale VERSION 0.1.0 LANGUAGES C CXX)

if(NOT WIN32)
    message(FATAL_ERROR "This CMake build is for Windows. Use the existing Makefile on macOS.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
option(STALE_BUILD_DESKTOP "Build the Slint Material desktop application" ON)
set(STALE_SLINT_SOURCE_DIR "" CACHE PATH "Source tree of the pinned jethac/slint fork")
set(STALE_SLINT_PACKAGE_DIR "" CACHE PATH "Optional installed CMake package of the same pinned fork")
set(STALE_SLINT_REV "46f3b237fc6a746356e9cd5535601b6d799637c0")

if(MSVC)
    add_compile_options(/utf-8 /EHsc)
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)

if(STALE_BUILD_DESKTOP)
    if(NOT STALE_SLINT_SOURCE_DIR)
        include(FetchContent)
        FetchContent_Declare(stale_slint
            URL "https://github.com/jethac/slint/archive/${STALE_SLINT_REV}.tar.gz"
            DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
        FetchContent_GetProperties(stale_slint)
        if(NOT stale_slint_POPULATED)
            FetchContent_Populate(stale_slint)
        endif()
        set(STALE_SLINT_SOURCE_DIR "${stale_slint_SOURCE_DIR}" CACHE PATH "Source tree of the pinned jethac/slint fork" FORCE)
    endif()
    # Compile the fork's runtime and compiler together. Embed Material fonts and
    # images so the executable does not depend on the build machine's paths.
    set(DEFAULT_SLINT_EMBED_RESOURCES "embed-files" CACHE STRING "" FORCE)
    set(SLINT_FEATURE_BACKEND_QT OFF CACHE BOOL "" FORCE)
    set(SLINT_FEATURE_RENDERER_SKIA OFF CACHE BOOL "" FORCE)
    set(SLINT_FEATURE_RENDERER_FEMTOVG ON CACHE BOOL "" FORCE)
    set(SLINT_FEATURE_RENDERER_SOFTWARE ON CACHE BOOL "" FORCE)
    set(SLINT_FEATURE_INTERPRETER OFF CACHE BOOL "" FORCE)
    set(SLINT_BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(SLINT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    if(STALE_SLINT_PACKAGE_DIR)
        find_package(Slint CONFIG REQUIRED PATHS "${STALE_SLINT_PACKAGE_DIR}" NO_DEFAULT_PATH)
    else()
        add_subdirectory("${STALE_SLINT_SOURCE_DIR}/api/cpp" "slint")
    endif()
endif()

add_subdirectory(windows)
if(STALE_BUILD_DESKTOP)
    add_subdirectory(desktop)
endif()
