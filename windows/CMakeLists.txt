cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(stale_windows LANGUAGES CXX)
  include(CTest)
endif()
if(NOT WIN32)
  message(FATAL_ERROR "The Stale Windows engine requires Windows")
endif()
add_library(stale_windows_core core.cpp browser.cpp)
target_include_directories(stale_windows_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(stale_windows_core PUBLIC cxx_std_17)
target_compile_definitions(stale_windows_core PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(stale_windows_core PUBLIC shell32 ole32 uuid bcrypt)
add_executable(stale_windows_cli cli.cpp)
target_link_libraries(stale_windows_cli PRIVATE stale_windows_core)
if(MSVC)
  target_compile_options(stale_windows_core PRIVATE /W4 /utf-8 /EHsc)
  target_compile_options(stale_windows_cli PRIVATE /W4 /utf-8 /EHsc)
endif()
if(BUILD_TESTING)
  add_executable(stale_windows_tests ../tests/windows/core_tests.cpp)
  target_link_libraries(stale_windows_tests PRIVATE stale_windows_core advapi32)
  if(MSVC)
    target_compile_options(stale_windows_tests PRIVATE /utf-8 /EHsc)
  endif()
  add_test(NAME stale_windows_core COMMAND stale_windows_tests)
  add_executable(stale_browser_tests ../tests/windows/browser_tests.cpp)
  target_include_directories(stale_browser_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
  target_link_libraries(stale_browser_tests PRIVATE stale_windows_core advapi32)
  add_test(NAME stale_windows_browser COMMAND stale_browser_tests)
endif()
